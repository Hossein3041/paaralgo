import java.util.concurrent.CyclicBarrier;
import java.util.concurrent.BrokenBarrierException;
class BitonicSortThread  extends Thread{
    int process_ID;
    int my_Value;
    int m_Stages;
    CyclicBarrier m_Barrier;

    private static final int[] mailbox = new int[8];

    public BitonicSortThread(int processID, int myValue, int stages, CyclicBarrier barrier){
        this.process_ID = processID;
        this.my_Value = myValue;
        this.m_Stages = stages;
        this.m_Barrier = barrier;
    }

    public synchronized void sendValueToPartner(int value, int partnerID){
        mailbox[partnerID] = value;
    }

    public synchronized int getValueFromPartner(int partnerID){
        return mailbox[process_ID];
    }

    public synchronized int getMin(int valA, int valB){
        return (valA < valB) ? valA : valB;
    } 

    public synchronized int getMax(int valA, int valB){
        return (valA > valB) ? valA : valB;
    }

    @Override
    public void run(){
        for(int dimension = m_Stages; dimension >= 1; --dimension){
            
            int partnerProcess_ID = process_ID ^ (1 << (dimension-1));
            boolean takeMin = (process_ID < (1 << (dimension - 1)));

            sendValueToPartner(my_Value, partnerProcess_ID);

            try{
                m_Barrier.await();
            } catch(InterruptedException | BrokenBarrierException e){
                Thread.currentThread().interrupt();
                System.err.println("Thread " + process_ID + " Synchronosationsfehler.");
            }

            int partner_Value = getValueFromPartner(partnerProcess_ID);

            if(takeMin)
                my_Value = getMin(my_Value, partner_Value);
            else
                my_Value = getMax(my_Value, partner_Value);

            try{
                m_Barrier.await();
            } catch(InterruptedException | BrokenBarrierException e){
                Thread.currentThread().interrupt();
                System.err.println("Thread " + process_ID + " Synchronisationsfehler.");
            }
        }
    }

    public synchronized int getSortedValue(){
        return my_Value;
    }
}
